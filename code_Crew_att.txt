            </div>

## components/FlightCard.js
```jsx
import React from 'react';
import './FlightCard.css';

const FlightCard = ({ flight }) => {
  return (
    <div className="flight-card">
      <div className="flight-info">
        <div className="departure-time">Departure {flight.departure}</div>
        <div className="flight-route">
          <span className="flight-number">{flight.flightNumber}</span>
          <span className="route"> • {flight.route}</span>
        </div>
        <div className="gate">{flight.gate}</div>
      </div>

      <div className="crew-info">
        <div className="crew-list">
          {flight.crew.map((member, index) => (
            <div key={index} className="crew-member">
              <img src={member.avatar} alt={member.name} className="avatar" />
              <span>Crew: {member.name}</span>
            </div>
          ))}
        </div>
      </div>

      <div className="contacts-info">
        <div className="contacts-title">Contacts</div>
        {flight.contacts.map((contact, index) => (
          <div key={index} className="contact-item">
            <strong>{contact.name}:</strong> {contact.phone}
          </div>
        ))}
      </div>

      <div className="attendance-info">
        <div className="attendance-title">Attendance</div>
        <div className={`attendance-status status-badge status-${flight.attendance.type}`}>
          {flight.attendance.status}
        </div>
        {flight.attendance.docs && (
          <div className="docs-status">Docs: {flight.attendance.docs}</div>
        )}
        {flight.attendance.nextCrew && (
          <div className="next-crew">Next available Crew: {flight.attendance.nextCrew}</div>
        )}
      </div>
    </div>
  );
};

export default FlightCard;
```

1. Create a new React app: `npx create-react-app flightops-dashboard`
2. Replace the default files with the code above
3. Add placeholder avatar images in the `public` folder or use the placeholder API URLs as shown
4. Run `npm start` to launch the application
